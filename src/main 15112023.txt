void setup()
{
#ifdef DEBUG
  Serial1.begin(115200);
#endif
  pinMode(ledPin, OUTPUT);
  Wire.begin();
  Wire.setClock(400000);
#ifdef DEBUG
  delay(1000);
#endif
  Serial1.println("IVY TECHNOLOGY - E2E");
  Serial1.println("Touchscreen tester V.1.0.6");

  TP_Detect();
  Serial1.println("DETECT");

  if (addr != 0)
  {
    digitalWrite(ledPin, HIGH);
  }
  TouchscreenUSB.begin();
  Serial1.println("TOUCH ON");
}

void loop()
{
  if (addr == 0x0A)
  {
    ReadTouchscreen_0A();
  } // RAMIA / BUMBLEBEE / ZIG-ZAG
  if (addr == 0x4B)
  {
    ReadTouchscreen_4B();
  } // FAROE
  if (addr == 0x10)
  {
    ReadTouchscreen_10();
  } //
}

//-----------------------------------------------------------------------------------------------------------------------------------
void TP_Detect()
{

  uint8_t i = 0;
  //*************************************************                              0x4B                       *************************************************
  Wire.requestFrom(0x4B, 3);
  while (Wire.available())
  {
    c[0] = Wire.read(); // 20
    c[1] = Wire.read(); // 0
    c[2] = Wire.read(); // 1
  }
  if (c[0] == 20 and c[1] == 0 and c[2] == 1)
    addr = 0x4B;
  //*************************************************   0x0A RAMIA / BUMBLEBEE / ZIG-ZAG / BERLINETTA CFL-R   *************************************************
  i = 0;
  Wire.beginTransmission(0x0A);
  Wire.write(0x00);
  Wire.write(0x00);
  Wire.endTransmission();
  // Read TS data: 30Bytes from 0x0A SLA ADDR
  Wire.requestFrom(0x0A, 30);

  while (Wire.available())
  {
    c[i] = Wire.read();
    i++;
  }

  // Ramia
  if (c[20] == 0x6A and c[21] == 0x05 and c[22] == 0x1C and c[23] == 0x48)
    addr = 0x0A;
  // Bumblebee
  else if (c[20] == 0x6A and c[21] == 0x05 and c[22] == 0x98 and c[23] == 0x48)
    addr = 0x0A;
  // ZigZag
  else if (c[20] == 0x6A and c[21] == 0x05 and c[22] == 0xD4 and c[23] == 0x48)
    addr = 0x0A;
  // Berlinetta
  else if (c[20] == 0x6A and c[21] == 0x05 and c[22] == 0x0B and c[23] == 0x49)
    addr = 0x0A;
  // Southpeak 13 (tgl)
  else if (c[20] == 0x6A and c[21] == 0x05 and c[22] == 0xA0 and c[23] == 0x49)
    addr = 0x0A;
  else
    addr = 0x0A;

  //*************************************************                 0x10 FAROE / MockingBird                  *************************************************
  i = 0;
  Wire.beginTransmission(0x10);
  Wire.write(01);
  Wire.write(00);
  Wire.endTransmission();

  Wire.requestFrom(0x10, 30);
  while (Wire.available())
  {
    c[i] = Wire.read();
    i++;
  }
  if (c[20] == 0xF3 and c[21] == 0x04 and c[22] == 0x82 and c[23] == 0x2A) // if (c[20] == 243 and c[21] == 4 and c[22] == 130 and c[23] == 42)
  {
    addr = 0x10;
#ifdef DEBUG
    Serial1.println("MOCKINGBIRD");
#endif
    // panel resolution: 1920x1080
    multi_x = 92;
    multi_y = 158;
  }
  else if (c[20] == 0xF3 and c[21] == 0x04 and c[22] == 0xE4 and c[23] == 0x29) // if (c[20] == 243 and c[21] == 4 and c[22] == 228 and c[23] == 41)
  {
    addr = 0x10;
#ifdef DEBUG
    Serial1.println("FIORANO");
#endif
    // panel resolution: 3840x2400
    multi_x = 84;
    multi_y = 134;
  }
  else if (c[20] == 0xF3 and c[21] == 0x04 and c[22] == 0xB4 and c[23] == 0x2E) // if (c[20] == 243 and c[21] == 4 and c[22] == 228 and c[23] == 41)
  {
    addr = 0x10;
#ifdef DEBUG
    Serial1.println("CINAMONBAY-16");
#endif
    // panel resolution: 3840x2400
    multi_x = 81;
    multi_y = 134;
  }
}

//-----------------------------------------------------------------------------------------------------------------------------------
void ReadTouchscreen_4B()
{
  uint8_t i = 0;
  int val = 0;
  // Read TouchScreen at 0x4B
  delay(5);
  Wire.requestFrom(0x4B, 20);
  while (Wire.available())
  {
    c[i] = Wire.read();
    i++;
  }

  reg = (c[18] * 256) + c[17];
  if (c[3] == 1 and c[19] == 1)
  {
    TouchscreenUSB.press(c[4], ((c[6] * 256) + c[5]) * 8, ((c[10] * 256) + c[9]) * 8);
    finger[c[4]] = 1;
  }
  else if (c[3] == 1 and c[19] >> 1)
  {
    TouchscreenUSB.press(c[4], ((c[6] * 256) + c[5]) * 8, ((c[10] * 256) + c[9]) * 8);
    finger[c[4]] = 1;
    val = c[19];
    for (i = 2; i = val; i++)
    {
      delay(5);

      Wire.requestFrom(0x4B, 20);
      i = 0;
      while (Wire.available())
      {
        c[i] = Wire.read();
        i++;
      }
      if (reg == (c[18] * 256) + c[17])
      {
        TouchscreenUSB.press(c[4], ((c[6] * 256) + c[5]) * 8, ((c[10] * 256) + c[9]) * 8);
        finger[c[4]] = 1;
      }
    }
  }
  else if (c[3] == 0)
  {
    TouchscreenUSB.release(0);
    finger[0] = 0;
    TouchscreenUSB.release(1);
    finger[1] = 0;
    TouchscreenUSB.release(2);
    finger[2] = 0;
    TouchscreenUSB.release(3);
    finger[3] = 0;
    TouchscreenUSB.release(4);
    finger[4] = 0;
    TouchscreenUSB.release(5);
    finger[5] = 0;
    TouchscreenUSB.release(6);
    finger[6] = 0;
    TouchscreenUSB.release(7);
    finger[7] = 0;
    TouchscreenUSB.release(8);
    finger[8] = 0;
    TouchscreenUSB.release(9);
    finger[9] = 0;
  }
  else if (c[19] != 0)
  {
    for (i = 0; i <= 9; i++)
    {
      if (finger[i] == 0)
        TouchscreenUSB.release(i);
      else
        finger[i] = 0;
    }
  }
}

void ReadTouchscreen_0A()
{ // RAMIA / BUMBLEBEE/ZIG-ZAG
  uint8_t i = 0;
  Wire.beginTransmission(0x0A);
  Wire.write(03);
  Wire.write(00);
  Wire.endTransmission();
  delay(1);

  Wire.requestFrom(0x0A, 40);
  while (Wire.available())
  {
    c[i] = Wire.read();
    i++;
  }
  if (c[0] == 0)
  {
    Wire.beginTransmission(0x0A);
    Wire.write(04);
    Wire.write(00);
    Wire.endTransmission();
    delay(1);

    Wire.requestFrom(0x0A, 40);
    i = 0;
    while (Wire.available())
    {

      c[i] = Wire.read();
      i++;
    }
    if (c[0] == 0)
    {
      for (i = 1; i <= 39; i++)
        c[i] = 0;
    }
  }

  if (c[4] >= 6)
  {
    do
    {
      Wire.beginTransmission(0x0A);
      Wire.write(03);
      Wire.write(00);
      Wire.endTransmission();
      delay(1);

      Wire.requestFrom(0x0A, 40);
      i = 0;
      while (Wire.available())
      {
        c[i] = Wire.read();
        i++;
        if (i == 4)
          i = 75;
        if (i == 76)
          i = 40;
      }
      if (c[0] == 0)
      {
        Wire.beginTransmission(0x0A);
        Wire.write(04);
        Wire.write(00);
        Wire.endTransmission();
        delay(1);

        Wire.requestFrom(0x0A, 40);
        i = 0;
        while (Wire.available())
        {
          //*******************************************************************************
          c[i] = Wire.read();
          i++;
          if (i == 4)
            i = 75;
          if (i == 76)
            i = 40;
        }
      }
    } while (c[75] != 0);
  }
  else
  {
    for (i = 40; i <= 75; i++)
    {
      c[i] = 0;
    }
  }

  for (i = 6; i <= 69; i = i + 7)
  {
    if (c[i] == 1)
    {
      TouchscreenUSB.press(0, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[0] = 1;
    }
    else if (c[i] == 2)
    {
      TouchscreenUSB.press(1, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[1] = 1;
    }
    else if (c[i] == 3)
    {
      TouchscreenUSB.press(2, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[2] = 1;
    }
    else if (c[i] == 4)
    {
      TouchscreenUSB.press(3, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[3] = 1;
    }
    else if (c[i] == 5)
    {
      TouchscreenUSB.press(4, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[4] = 1;
    }
    else if (c[i] == 6)
    {
      TouchscreenUSB.press(5, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[5] = 1;
    }
    else if (c[i] == 7)
    {
      TouchscreenUSB.press(6, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[6] = 1;
    }
    else if (c[i] == 8)
    {
      TouchscreenUSB.press(7, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[7] = 1;
    }
    else if (c[i] == 9)
    {
      TouchscreenUSB.press(8, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[8] = 1;
    }
    else if (c[i] == 10)
    {
      TouchscreenUSB.press(9, ((c[i + 3] * 256) + c[i + 2]) * 2.8, ((c[i + 5] * 256) + c[i + 4]) * 5);
      finger[9] = 1;
    }
  }

  for (i = 0; i <= 9; i++)
  {
    if (finger[i] == 0)
      TouchscreenUSB.release(i);
    else
      finger[i] = 0;
  }
  delay(4);
}
