/*
 *
 * I2C slave address scan:
 * result I2C address array and number of detected devices
 *
 */

void SlaAddrDetect()
{
  int error = 0;
  nDevices = 0;
  for (i = 1; i < 128; i++)
  {
    Wire.beginTransmission(i);
    error = Wire.endTransmission();
    if (error == 0 && i != 0 && i != 0x78 && i != 0x77)
    {
      dev_addr[nDevices] = i;
      nDevices++;
    }
  }
  for (i = 0; i < nDevices; i++)
  {
    if (dev_addr[i] == 0x10)
      addr = 0x10;
    else if (dev_addr[i] == 0x0A)
      addr = 0x0A;
    else if (dev_addr[i] == 0x4B)
      addr = 0x4B;
    else if (dev_addr[i] == 0x34)
      addr = 0x34;
  }
#ifdef DEBUG1
  Serial1.print("Found: ");
  Serial1.print(nDevices, DEC);
  Serial1.println(" devices");
  for (i = 0; i < nDevices; i++)
  {
    if (dev_addr[i] < 16)
      Serial1.print("0x0");
    else
      Serial1.print("0x");
    Serial1.print(dev_addr[i], HEX);
    Serial1.print(", ");
  }
  Serial1.println("");
#endif
}

/* *************************************************************************************************************************************************************
 * check HID frame 30 bytes: read from SLA_ADDR, REG_ADDR_L (0x00), REG_ADDR_H(0x00)
 * e.g.
 *
 *    1E	0 	0 	1	 31 	2 	2 	0 	3 	0 	43	0 	4 	0 	FF	0	  5	  0	  6  	0 	F3	 4	 E4 	29 	5	 61 	0 	0 	0 	0
 * data bytes order: data byte low, data byte high
 * 16bit value: data byte high * 256 + data byte low
 *
 * 2 bytes: HID DESC LENGTH
 * 2 bytes: BCD VERSION
 * 2 bytes: REPORT DESC LENGTH
 * 2 bytes: REPORT REGISTER
 * 2 bytes: INPUT REGISTER
 * 2 bytes: MAX INPUT LENGTH
 * 2 bytes: OUTPUT REGISTER
 * 2 bytes: MAX OUTPUT LENGTH
 * 2 bytes: COMMAND REGISTER
 * 2 bytes: DATA REGISTER
 * 2 bytes: VENDOR ID
 * 2 bytes: PRODUCT ID
 * 2 bytes: VERSION ID
 * 4 bytes: RESERVED
 */

void TP_Detect()
{
  uint8_t error = 0;

  Serial1.println("....detect");
  uint8_t i = 0;

  if (addr == 0x34)
  {
    Wire.beginTransmission(0x34);
    Wire.write(00);
    Wire.write(00);
    Wire.endTransmission();

    Wire.requestFrom(0x34, 30);
    while (Wire.available())
    {
      data[i] = Wire.read();
      i++;
    }
    if (data[20] == 0xD2 and data[21] == 0x1F and data[22] == 0x05 and data[23] == 0xB0)
    {
      addr = 0x34;
#ifdef DEBUG_IN
      Serial1.println("SOUTHPEAK 14/V50T5");
#endif
      // panel resolution: 1920x1080
      // multi_x = 96;
      // multi_y = 162;
      multi_x = 80;
      multi_y = 79;
    }
  }
  //*************************************************                           FAROE                         *************************************************
  if (addr == 0x4B)
  {
    Wire.requestFrom(0x4B, 3);
    while (Wire.available())
    {
      data[0] = Wire.read(); // 20
      data[1] = Wire.read(); // 0
      data[2] = Wire.read(); // 1
    }
    if (data[0] == 20 and data[1] == 0 and data[2] == 1)
      addr = 0x4B;
  }
  //*************************************************   0x0A RAMIA / BUMBLEBEE / ZIG-ZAG / BERLINETTA CFL-R   *************************************************
  if (addr == 0x0A)
  {
    i = 0;
    Wire.beginTransmission(0x0A);
    Wire.write(0x00);
    Wire.write(0x00);
    error = Wire.endTransmission();
    Wire.requestFrom(0x0A, 30); // Read TS data: 30Bytes from 0x0A SLA ADDR
    while (Wire.available())
    {
      data[i] = Wire.read();
      i++;
    }

    if (data[20] == 0x6A and data[21] == 0x05 and data[22] == 0x1C and data[23] == 0x48)
    {
#ifdef DEBUG
      Serial1.println("RAMIA");
#endif
      addr = 0x0A;
    }
    else if (data[20] == 0x6A and data[21] == 0x05 and data[22] == 0x98 and data[23] == 0x48)
    {
#ifdef DEBUG
      Serial1.println("BUMBLEBEE");
#endif
      addr = 0x0A;
    }
    else if (data[20] == 0x6A and data[21] == 0x05 and data[22] == 0xD4 and data[23] == 0x48)
    {
#ifdef DEBUG
      Serial1.println("ZIG_ZAG");
#endif
      addr = 0x0A;
    }
    else if (data[20] == 0x6A and data[21] == 0x05 and data[22] == 0x0B and data[23] == 0x49)
    {
#ifdef DEBUG
      Serial1.println("BERLINETTA");
#endif
      addr = 0x0A;
    }
    else if (data[20] == 0x6A and data[21] == 0x05 and data[22] == 0xA0 and data[23] == 0x49)
    {
#ifdef DEBUG
      Serial1.println("SOUTHPEAK 13 TGL");
#endif
      Southpeak_TS_Init();
      platform = SOUTHPEAK_13_TGL;
      addr = 0x0A;
    }
  }
  //*************************************************                      ELAN                    *************************************************
  if (addr == 0x10)
  {

    i = 0;
    Wire.beginTransmission(0x10);
    Wire.write(01);
    Wire.write(00);
    Wire.endTransmission();

    Wire.requestFrom(0x10, 30);
    while (Wire.available())
    {
      data[i] = Wire.read();
      i++;
    }
    if (data[20] == 0xF3 and data[21] == 0x04 and data[22] == 0x82 and data[23] == 0x2A) // if (data[20] == 243 and data[21] == 4 and data[22] == 130 and data[23] == 42)
    {
      addr = 0x10;
#ifdef DEBUG_IN
      Serial1.println("MOCKINGBIRD");
#endif
      // panel resolution: 1920x1080
      multi_x = 92;
      multi_y = 158;
    }
    else if (data[20] == 0xF3 and data[21] == 0x04 and data[22] == 0xE4 and data[23] == 0x29)
    {
      addr = 0x10;
#ifdef DEBUG_IN
      Serial1.println("FIORANO");
#endif
      // panel resolution: 3840x2400
      multi_x = 84;
      multi_y = 134;
    }
    else if (data[20] == 0xF3 and data[21] == 0x04 and data[22] == 0xB4 and data[23] == 0x2E)
    {
      addr = 0x10;
#ifdef DEBUG_IN
      Serial1.println("CINAMONBAY-16");
#endif
      // panel resolution: 3840x2400
      multi_x = 81;
      multi_y = 134;
    }
    else if (data[20] == 0xF3 and data[21] == 0x04 and data[22] == 0x46 and data[23] == 0x28)
    {
      addr = 0x10;
#ifdef DEBUG_IN
      Serial1.println("SOUTHPEAK 14 MLK");
#endif
      // panel resolution: 1920x1080
      multi_x = 93;
      multi_y = 164;
    }
    else if (data[20] == 0xF3 and data[21] == 0x04 and data[22] == 0x56 and data[23] == 0x28)
    {
      addr = 0x10;
#ifdef DEBUG_IN
      Serial1.println("SOUTHPEAK 14");
#endif
      // panel resolution: 1920x1080
      multi_x = 99;
      multi_y = 173;
    }
    else if (data[20] == 0xF3 and data[21] == 0x04 and data[22] == 0x85 and data[23] == 0x28)
    {
      addr = 0x10;
#ifdef DEBUG_IN
      Serial1.println("NORTH BAY 15P MLK");
#endif
      // panel resolution: 1920x1080
      multi_x = 84;
      multi_y = 152;
    }
    else if (data[20] == 0xF3 and data[21] == 0x04 and data[22] == 0x30 and data[23] == 0x28)
    {
      addr = 0x10;
#ifdef DEBUG_IN
      Serial1.println("SOUTHPEAK 13");
#endif
      // panel resolution: 1920x1080
      multi_x = 96;
      multi_y = 162;
    }
    else if (data[20] == 0xF3 and data[21] == 0x04 and data[22] == 0xA7 and data[23] == 0x2B)
    {
      addr = 0x10;
#ifdef DEBUG_IN
      Serial1.println("SOUTHPEAK 14/ PANEL JN7TT");
#endif

    i = 0;

void ReadTouchscreen_4B()
{
  uint8_t i = 0;
  int val = 0;
  // Read TouchScreen at 0x4B
  delay(5);
  Wire.requestFrom(0x4B, 20);
  while (Wire.available())
  {
    data[i] = Wire.read();
    i++;
  }

  reg = (data[18] * 256) + data[17];
  if (data[3] == 1 and data[19] == 1)
  {
    TouchscreenUSB.press(data[4], ((data[6] * 256) + data[5]) * 8, ((data[10] * 256) + data[9]) * 8);
    finger[data[4]] = 1;
  }
  else if (data[3] == 1 and data[19] >> 1)
  {
    TouchscreenUSB.press(data[4], ((data[6] * 256) + data[5]) * 8, ((data[10] * 256) + data[9]) * 8);
    finger[data[4]] = 1;
    val = data[19];
    for (i = 2; i = val; i++)
    {
      delay(5);

      Wire.requestFrom(0x4B, 20);
      i = 0;
      while (Wire.available())
      {
        data[i] = Wire.read();
        i++;
      }
      if (reg == (data[18] * 256) + data[17])
      {
        TouchscreenUSB.press(data[4], ((data[6] * 256) + data[5]) * 8, ((data[10] * 256) + data[9]) * 8);
        finger[data[4]] = 1;
      }
    }
  }
  else if (data[3] == 0)
  {
    for (int index = 0; index < 10; index++)
    {
      TouchscreenUSB.release(i);
      finger[i] = 0;
    }
  }
  else if (data[19] != 0)
  {
    for (i = 0; i <= 9; i++)
    {
      if (finger[i] == 0)
        TouchscreenUSB.release(i);
      else
        finger[i] = 0;
    }
  }
}

void ReadTouchscreen_0A()
{ // RAMIA / BUMBLEBEE/ZIG-ZAG
  uint8_t i = 0;
  Wire.beginTransmission(0x0A);
  Wire.write(03);
  Wire.write(00);
  Wire.endTransmission();
  delay(1);

  Wire.requestFrom(0x0A, 40);
  while (Wire.available())
  {
    data[i] = Wire.read();
    i++;
  }
  if (data[0] == 0)
  {
    Wire.beginTransmission(0x0A);
    Wire.write(04);
    Wire.write(00);
    Wire.endTransmission();
    delay(1);

    Wire.requestFrom(0x0A, 40);
    i = 0;
    while (Wire.available())
    {

      data[i] = Wire.read();
      i++;
    }
    if (data[0] == 0)
    {
      for (i = 1; i <= 39; i++)
        data[i] = 0;
    }
  }

  if (data[4] >= 6)
  {
    do
    {
      Wire.beginTransmission(0x0A);
      Wire.write(03);
      Wire.write(00);
      Wire.endTransmission();
      delay(1);

      Wire.requestFrom(0x0A, 40);
      i = 0;
      while (Wire.available())
      {
        data[i] = Wire.read();
        i++;
        if (i == 4)
          i = 75;
        if (i == 76)
          i = 40;
      }
      if (data[0] == 0)
      {
        Wire.beginTransmission(0x0A);
        Wire.write(04);
        Wire.write(00);
        Wire.endTransmission();
        delay(1);

        Wire.requestFrom(0x0A, 40);
        i = 0;
        while (Wire.available())
        {
          //*******************************************************************************
          data[i] = Wire.read();
          i++;
          if (i == 4)
            i = 75;
          if (i == 76)
            i = 40;
        }
      }
    } while (data[75] != 0);
  }
  else
  {
    for (i = 40; i <= 75; i++)
    {
      data[i] = 0;
    }
  }

  for (i = 6; i <= 69; i = i + 7)
  {
    if (data[i] == 1)
    {
      TouchscreenUSB.press(0, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[0] = 1;
    }
    else if (data[i] == 2)
    {
      TouchscreenUSB.press(1, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[1] = 1;
    }
    else if (data[i] == 3)
    {
      TouchscreenUSB.press(2, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[2] = 1;
    }
    else if (data[i] == 4)
    {
      TouchscreenUSB.press(3, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[3] = 1;
    }
    else if (data[i] == 5)
    {
      TouchscreenUSB.press(4, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[4] = 1;
    }
    else if (data[i] == 6)
    {
      TouchscreenUSB.press(5, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[5] = 1;
    }
    else if (data[i] == 7)
    {
      TouchscreenUSB.press(6, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[6] = 1;
    }
    else if (data[i] == 8)
    {
      TouchscreenUSB.press(7, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[7] = 1;
    }
    else if (data[i] == 9)
    {
      TouchscreenUSB.press(8, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[8] = 1;
    }
    else if (data[i] == 10)

      TouchscreenUSB.press(9, ((data[i + 3] * 256) + data[i + 2]) * 2.8, ((data[i + 5] * 256) + data[i + 4]) * 5);
      finger[9] = 1;
    }

  for (i = 0; i <= 9; i++)
  {
    if (finger[i] == 0)
      TouchscreenUSB.release(i);
    else
      finger[i] = 0;
  }
  delay(4);
}

void Southpeak_TS_Init()
{
  Wire.beginTransmission(0x0A);
  Wire.write(0x04);
  Wire.write(0x00);
  Wire.write(0x00);
  Wire.write(0x08);
  Wire.endTransmission();
  delay(2);
  Wire.beginTransmission(0x0A);
  Wire.write(0x04);
  Wire.write(0x00);
  Wire.write(0x00);
  Wire.write(0x01);
  Wire.endTransmission();
  delay(30);
  Wire.beginTransmission(0x0A);
  Wire.write(0x04);
  Wire.write(0x00);
  Wire.write(0x1C);
  Wire.write(0x04);
  Wire.endTransmission();
}

/* for mockingbird, before use ReadTouchscreen_Mockingbird_13_TGL(), please use Southpeak_TS_Init()
 * after initialization , TS controller sends 1-5 24-bits data frames,
 * each dataframe  contains two finger coordinates
 *
 * for 1-2 fingers active controller sends one data frame
 * * for 3-4 fingers active controller sends two data frames
 * * * for 5-6 fingers active controller sends three data frames
 * * * * for 7-8 fingers active controller sends four data frames
 * * * * * for 9-10 fingers active controller sends five data frames
 * e.g 10 fingers detected
 *
 *         15	00	1C	00	0A	05	01	00	18	06	E2	0C	05	02	00	AC	23	99	05	AB	2F	00	00	00
 *         15	00	1C	00	00	05	03	00	B6	0A	24	09	05	04	00	D2	27	3C	08	AB	2F	00	00	00
 *         15	00	1C	00	00	05	05	00	17	1F	22	05	05	06	00	FC	13	64	15	AB	2F	00	00	00
 *         15	00	1C	00	00	05	07	00	48	1A	E0	14	05	08	00	5A	1A	1B	09	AB	2F	00	00	00
 *         15	00	1C	00	00	05	09	00	22	0E	C6	06	05	0A	00	A8	12	13	0A	AB	2F	00	00	00
 *
 * 5 byte - number of detected fingers
 * 6, 13 byte - finger number
 * 9-10, 16-17 bytes - x position
 * 11-12, 18-19 bytes - y position
 *
 * only first frame contains a field informing about the number of detected fingers
 * in other frames this field always contains the value 0x00
 * if second finger in frame is not detected, bytes 13-19 always equal 0x00
 * if pair of fingers is not detected - TS controller skips sending this frame
 * if any finger is not detected, - TS controller sends frame 24-bytes frame 0x04, 0x00, 0xFF.....0xFF
 *
 */

void ReadTouchscreen_Southpeak_13_TGL()
{

  int mult_x = 1;
  int mult_y = 1;
  int fingers = 0;

  ReadTouchscreen(0x0A, data, 24, 0);
  //  check how many fingers is detected
  if (data[4] != 0x00 && data[4] != 0xFF)
    fingers = data[4];

  if (fingers <= 4)
  {
    delayMicroseconds(1200);
    ReadTouchscreen(0x0A, data, 24, 24);
  }
  else if (fingers <= 6)
  {
    delayMicroseconds(1200);
    ReadTouchscreen(0x0A, data, 24, 48);
  }

    delayMicroseconds(1200);

  }

  // if (data[6] = 2)

  // if (data[6] = 3)
  //   TouchscreenUSB.press(0, ((data[24 + 9] * 256) + data[24 + 8]) * mult_x, ((data[24 + 12] * 256) + data[24 + 11]) * mult_y);
  // if (data[6] = 4)

  // if (data[6] = 5)
  //   TouchscreenUSB.press(0, ((data[48 + 9] * 256) + data[48 + 8]) * mult_x, ((data[48 + 12] * 256) + data[48 + 11]) * mult_y);
  // if (data[6] = 6)
  //   TouchscreenUSB.press(0, ((data[48 + 17] * 256) + data[48 + 16]) * mult_x, ((data[48 + 19] * 256) + data[48 + 18]) * mult_y);
  // if (data[6] = 7)

//                  (slave_address, pointer_to_data_array, array_size, index)
void ReadTouchscreen(int slaAddr, int *pArray, int size, int index)
{
  //  int index = 0;
  Wire.requestFrom(slaAddr, size);
  while (Wire.available())
  {
    pArray[index] = Wire.read();
    index++;
  }
}

void print_data_table(int *pArray, int size)
{
  for (i = 0; i <= size; i++)
  {
    Serial.print(pArray[i], HEX);
    Serial.print(",");
  }
  Serial.println("");
}
